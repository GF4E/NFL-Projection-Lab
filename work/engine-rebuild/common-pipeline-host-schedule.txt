# /etc/systemd/system/nfl-engine-daily.timer
[Unit]
Description=NFL daily job, retaining hourly public-final refresh cadence

[Timer]
OnCalendar=hourly
Persistent=true
AccuracySec=30s
Unit=nfl-engine-daily.service

[Install]
WantedBy=timers.target
